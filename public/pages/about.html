// decideTurn runs once per turn and returns this turn's actions.
function decideTurn(ctx) {
  // Turn 1 you have no units — buy an army (costs money, not AP).
  if (ctx.turn === 1) {
    return [
      { action: "buy", unitType: "knight" },
      { action: "buy", unitType: "archer" },
      { action: "buy", unitType: "priest" },
    ];
  }

  const actions = [];
  for (const u of ctx.myUnits) {
    // Find the nearest enemy by Manhattan distance.
    let target = null, best = Infinity;
    for (const e of ctx.enemyUnits) {
      const d = Math.abs(u.pos[0] - e.pos[0]) + Math.abs(u.pos[1] - e.pos[1]);
      if (d < best) { best = d; target = e; }
    }
    if (!target) { actions.push({ unitId: u.id, action: "defend" }); continue; }

    // One action per unit: attack if close, else step toward it.
    if (best <= 2) {
      actions.push({ unitId: u.id, action: "attack", targetUnitId: target.id });
    } else {
      const nx = u.pos[0] + Math.sign(target.pos[0] - u.pos[0]);
      const ny = u.pos[1] + Math.sign(target.pos[1] - u.pos[1]);
      actions.push({ unitId: u.id, action: "move", target: [nx, ny] });
    }
  }
  return actions;
}
